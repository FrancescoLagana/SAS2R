Apply formats from metadata file

Published

November 6, 2026

1 Aim: Apply formats from metadata file to a data table dataset.

Load packages

library(data.table)

Formats are in a list of named vectors, where the names are the labels and the values are the codes.

# Create a mapping list for labels 
label_list <- list(
  sesso = c(
    "1" = "Maschio",
    "2" = "Femmina"
  ),
  istruzione = c(
    "1" = "Obbligatoria",
    "2" = "Secondaria",
    "3" = "Terziaria"
  )
)

Retrieve values from the list (equivalent of select statement in SAS PROC FORMAT) and apply them to the dataset. The haven::labelled() function is used to apply the labels to the variables.

4 Apply formats to data with haven package

Apply formats to data

# Invert mapping: code -> label becomes label -> code
label_list_haven <- lapply(label_list, function(x) {
  setNames(as.numeric(names(x)), unname(x))
})

# Apply labels on-the-fly using the mapping list
DT[, sesso := haven::labelled(sesso, labels = label_list_haven$sesso)]
DT[, istruzione := haven::labelled(istruzione, labels = label_list_haven$istruzione)]

  by = .(sesso = haven::as_factor(sesso),
         istruzione = haven::as_factor(istruzione))]

5 Apply formats without using haven package

library(data.table)

# Dataset
DT <- data.table(
  sesso = c(1, 2, 1, 2, 1, 2),
  istruzione = c(1, 2, 3, 2, 1, 3)
)

# Mapping dei metadati: codice -> etichetta
label_list <- list(
  sesso = c(
    "1" = "Maschio",
    "2" = "Femmina"
  ),
  istruzione = c(
    "1" = "Obbligatoria",
    "2" = "Secondaria",
    "3" = "Terziaria"
  )
)

Apply value labels as attributes to the dataset, with a loop

# Applicazione delle value labels come attributi
for (v in names(DT)) {
  setattr(DT[[v]], "labels", label_list[[v]])
}

… tabulate with labels

# Funzione per convertire i codici in etichette
as_label <- function(x, labels) {
  factor(
    as.character(x),
    levels = names(labels),
    labels = unname(labels)
  )
}

# Tabulazione con etichette
tab <- DT[, .(Count = .N),
  by = .(
    sesso = as_label(sesso, label_list$sesso),
    istruzione = as_label(istruzione, label_list$istruzione)
  )
]

# Risultato
